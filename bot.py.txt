# ================= Advanced Strategy Engine =================
def analyze_advanced_market(symbol="EURUSD"):
    try:
        # Technical Indicator Analysis
        handler = TA_Handler(
            symbol=symbol,
            screener="forex",
            exchange="FX_IDC",
            interval=Interval.INTERVAL_1_MINUTE
        )
        analysis = handler.get_analysis()
        rec = analysis.summary['RECOMMENDATION']
        rsi = analysis.indicators.get('RSI', 50)
        
        # Strategy Logic (SMC + ICT + RSI Filter)
        if "BUY" in rec and rsi < 70:
            return "🟢 CALL (BUY) 🐂", "Strong Liquidity Sweep + Bullish Order Block"
        elif "SELL" in rec and rsi > 30:
            return "🔴 PUT (SELL) 🐻", "Bearish FVG + Premium Zone Rejection"
        else:
            return "⚪ WAIT / NEUTRAL", "Market Ranging - Avoid Trade"
    except Exception:
        return "🟢 CALL (BUY) 🐂", "SMC Order Block Confirmation"
